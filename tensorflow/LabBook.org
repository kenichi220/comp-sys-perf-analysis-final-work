meu_estilo <- function() {
  list(
    theme_bw(base_size = 20),
    theme(
      legend.title = element_blank(),
      plot.margin = unit(c(0, 0, 0, 0), "cm"),
      legend.spacing = unit(1, "mm"),
      legend.position = "right",
      legend.justification = "left",
      legend.box.spacing = unit(0, "pt"),
      legend.box.margin = margin(0, 0, 0, 0),
      axis.text.x = element_text(angle=45, vjust=1, hjust=1)
    ))
}

  mutate(
    # 'epoca' must be numeric for plotting
    epoca = as.numeric(epoca),
    # 'tempo' must be numeric for plotting
    tempo = as.numeric(tempo),
    # Create a unique identifier for each configuration/run (e.g., combining ID and Configs)
    # Replace 'config_column_1' and 'config_column_2' with your actual column names.
    config_run = paste(model, batch, type, nodes, sep = "_")
  ) -> data_plot
data_plot |>
  group_by(config_run, epoca) |>
  summarise(
    N = n(),
    mean_tempo = mean(tempo, na.rm = TRUE),
    sd_tempo = sd(tempo),
    se_tempo = 3 * sd(tempo) / sqrt(N),
    model = model,
    .groups = 'drop'
  ) |>
  mutate(
    lower_bound = mean_tempo - se_tempo,
    upper_bound = mean_tempo + se_tempo
  ) |>
  ggplot(aes(x = epoca, y = mean_tempo, color = as.factor(epoca))) +

  geom_jitter(
    data = data_plot,
    aes(x = epoca, y = tempo, color = as.factor(epoca), group = config_run),
    # Set the transparency (alpha) to 0.3 for a faint look
    alpha = 0.3,
    size = 2
  ) +

  # Add points for the mean of the epochs
  geom_point(size = 3, alpha = 0) +
  # Add the error bars (using the calculated lower and upper bounds)
