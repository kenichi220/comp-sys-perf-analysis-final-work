  list(

get_data_comb <- function(df, mod, typ){

  ret <- df |> filter(grepl(mod, model)) |> filter(type == typ) |> group_by(id) |>

    mutate(nodes = as.numeric(nodes)) |>

    select(-epoca, -tempo) |>
