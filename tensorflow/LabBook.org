get_data_comb <- function(df, mod, typ){

  ret <- df |> filter(grepl(mod, model)) |> filter(type == typ) |> group_by(id) |>
    mutate(tempo_treino = sum(as.numeric(tempo))) |>
    mutate(nodes = as.factor(nodes)) |>
    mutate(batch = as.factor(batch)) |>
    select(-epoca, -tempo) |>
    ungroup()


  return(ret)
}

library(dplyr)
library(ggplot2)

# --- 1. Reestruturar Variáveis e Modelo ---
# O modelo categórico é o ajuste mais adequado
R$nodes_f <- factor(R$nodes)
R$batch_f <- factor(R$batch)
modelo_categorico <- lm(tempo_treino ~ nodes_f * batch_f, data = R)

# --- 2. Preparar Dados Agregados ---
dados_agregados <- R %>%
  group_by(nodes_f, batch_f) %>%
  summarise(
    # Média observada (ponto)
    media_tempo_observado = mean(tempo_treino),
    # Erro Padrão (para barras de erro)
    se_tempo = sd(tempo_treino) / sqrt(n()),
    .groups = 'drop'
  )

# 3. Adicionar Previsões do Modelo
# As previsões do modelo categórico são as médias dos grupos
dados_agregados$previsao_modelo <- predict(modelo_categorico, newdata = dados_agregados)

# --- 4. Criar o Gráfico Combinado ---
grafico_combinado <- dados_agregados %>%
  ggplot(aes(x = batch_f, group = nodes_f, color = nodes_f)) +

  # A. Pontos Observados (Média de cada Grupo)
  geom_point(aes(y = media_tempo_observado),
             size = 4,
             alpha = 0.8) +

  # B. Linhas do Modelo (Previsão para cada Grupo)
  geom_line(aes(y = previsao_modelo),
            linewidth = 1.2,
            linetype = "solid") +

  # C. Barras de Erro (Intervalo de Confiança 95% = Média +/- 1.96*SE)
  geom_errorbar(aes(ymin = media_tempo_observado - 1.96 * se_tempo,
                    ymax = media_tempo_observado + 1.96 * se_tempo),
                width = 0.15) +

  # Rótulos e Títulos
  labs(
    title = "Efeito de Interação (Modelo Categórico)",
    subtitle = "Linhas e Pontos representam as Médias Observadas e Previstas",
    x = "Tamanho do Batch",
    y = "Tempo de Treino Médio (segundos)",
    color = "Nodos"
  ) +
  theme_minimal()

# Exibir o gráfico
print(grafico_combinado)
